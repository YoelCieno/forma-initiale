**Goal:** Rename `apps/web-vue/` to `apps/white-label-vue/`, add `layer.config.ts`, split `main.ts` into factory + entry.

## Key concept

- `layer.config.ts` exports white-label's component dirs, route config, Vite extensions — consumed by tenant `vite.config.ts`
- `bootstrap/app.ts` exports `createWhiteLabelApp(options?)` — accepts route overrides, component dirs, token overrides
- `main.ts` calls factory with defaults — standalone dev mode unchanged
- Workspace glob `apps/*` handles rename with no turbo.json changes

  ```typescript
  import { createApp, type App as VueApp } from 'vue'
  import {
    createRouter,
    createWebHashHistory,
    type RouteRecordRaw,
  } from 'vue-router'
  import App from './App.vue'
  import { layerConfig } from '../layer.config'

  export interface WhiteLabelAppOptions {
    routes?: RouteRecordRaw[]
    mocks?: boolean
    element?: string
    beforeMount?: (app: VueApp) => void
  }

  export function createWhiteLabelApp(options: WhiteLabelAppOptions = {}) {
    const {
      routes = layerConfig.routes,
      mocks = false,
      element = '#app',
    } = options

    const app = createApp(App)
    const router = createRouter({
      history: createWebHashHistory(),
      routes,
    })

    app.use(router)

    if (options.beforeMount) {
      options.beforeMount(app)
    }

    return { app, router, element }
  }
  ```
